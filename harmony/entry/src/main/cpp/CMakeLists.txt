cmake_minimum_required(VERSION 3.28)
project(persona_ssh LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(persona_ssh SHARED ssh_napi.cpp)
add_library(persona_ssh_core SHARED IMPORTED)
if(DEFINED ENV{PERSONA_SSH_LIBRARY})
  set(PERSONA_SSH_LIBRARY "$ENV{PERSONA_SSH_LIBRARY}")
else()
  set(PERSONA_SSH_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native/ssh/target/aarch64-unknown-linux-ohos/release/libpersona_ssh_core.so")
endif()
if(NOT OHOS_ARCH STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Persona SSH currently requires the validated ARM64 target")
endif()
if(NOT EXISTS "${PERSONA_SSH_LIBRARY}")
  message(FATAL_ERROR "Build the locked Persona SSH Rust library before building the HAP")
endif()
set_target_properties(persona_ssh_core PROPERTIES IMPORTED_LOCATION "${PERSONA_SSH_LIBRARY}" IMPORTED_NO_SONAME TRUE)
target_link_libraries(persona_ssh PRIVATE libace_napi.z.so persona_ssh_core)
set_target_properties(persona_ssh PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN" BUILD_WITH_INSTALL_RPATH TRUE)
add_custom_command(TARGET persona_ssh POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  $<TARGET_FILE:persona_ssh_core> $<TARGET_FILE_DIR:persona_ssh>/libpersona_ssh_core.so)
